SELECT
  c.id   AS category_id,
  c.name AS category_name,
  m.id   AS module_id,
  m.name AS module_name
FROM modules c
LEFT JOIN modules m ON m.parent_id = c.id AND m.deleted_at IS NULL
WHERE c.is_core = true AND c.deleted_at IS NULL
ORDER BY c.name, m.name;
